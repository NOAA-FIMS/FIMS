677 fims::Vector<fims::Vector<double>> PrepareFleetObservationBinEdges(
678 const std::shared_ptr<fims_popdy::Population<Type>>& p) {
679 fims::Vector<fims::Vector<double>> fleet_edges;
680
681 for (std::size_t fleet_index = 0; fleet_index < p->fleets.size();
682 ++fleet_index) {
683 const std::shared_ptr<fims_popdy::Fleet<Type>>& fleet =
684 p->fleets[fleet_index];
685
686 if (fleet == nullptr) {
687 continue;
688 }
689
690 if (!fleet->requires_age_length_mapping || fleet->n_lengths == 0) {
691 fleet->length_bin_edges = fims::Vector<double>();
692 continue;
693 }
694
695 if (fleet->lengths.size() != fleet->n_lengths) {
696 throw std::runtime_error(
697 "Fleet length_bin centers are not consistent with n_lengths");
698 }
699
700 fleet->length_bin_edges =
701 fims_popdy::SizeGridBuilder::BuildObservationEdgesFromCenters(
702 fleet->lengths);
703
704 fleet_edges.emplace_back(fleet->length_bin_edges);
705 }
706
707 return fleet_edges;
708 }

723 void EnforcePopulationGridNotCoarserThanFleetBins(
724 const fims_popdy::SizeGrid& size_grid,
725 const fims::Vector<fims::Vector<double>>& fleet_edges) {
726 if (!size_grid.IsConsistent() || size_grid.n_bins == 0 ||
727 fleet_edges.size() == 0) {
728 return;
729 }
730
731 for (std::size_t population_bin_index = 0;
732 population_bin_index < size_grid.n_bins; ++population_bin_index) {
733 const double population_left = size_grid.edges[population_bin_index];
734 const double population_right = size_grid.edges[population_bin_index + 1];
735 const double population_bin_width = population_right - population_left;
736 const double population_tolerance =
737 1e-12 * (population_bin_width > 1.0 ? population_bin_width : 1.0);
738
739 for (std::size_t fleet_index = 0; fleet_index < fleet_edges.size();
740 ++fleet_index) {
741 const fims::Vector<double>& edges = fleet_edges[fleet_index];
742
743 for (std::size_t fleet_bin_index = 1; fleet_bin_index < edges.size();
744 ++fleet_bin_index) {
745 const double fleet_left = edges[fleet_bin_index - 1];
746 const double fleet_right = edges[fleet_bin_index];
747 const double overlap_left = std::max(population_left, fleet_left);
748 const double overlap_right = std::min(population_right, fleet_right);
749
750 if (!(overlap_right > overlap_left)) {
751 continue;
752 }
753
754 const double fleet_bin_width = fleet_right - fleet_left;
755 if (population_bin_width > fleet_bin_width + population_tolerance) {
756 throw std::runtime_error(
757 "Population biological size grid contains at least one bin "
758 "that is coarser than an overlapping active fleet observation "
759 "bin. Provide a finer biological size-grid override or correct "
760 "the fleet bin setup.");
761 }
762 }
763 }

792 FIMS_ERROR_LOG(
793 "Failed to prepare fleet observation-bin geometry for population " +
794 fims::to_string(p->id) + ": " + e.what());
795 return;
796 }
797
798 if (p->size_grid.IsConsistent() && p->size_grid.n_bins > 0) {
799 try {
800 EnforcePopulationGridNotCoarserThanFleetBins(p->size_grid, fleet_edges);
801 } catch (const std::exception& e) {
802 valid_model = false;
803 FIMS_ERROR_LOG(
804 "Population " + fims::to_string(p->id) +
805 " has a biological size grid that is incompatible with active "
806 "fleet observation bins: " +
807 e.what());
808 }
809 return;
810 }
811
812 const bool size_grid_missing = p->size_grid.n_bins == 0 &&
813 p->size_grid.edges.size() == 0 &&
814 p->size_grid.centers.size() == 0;
815
816 if (size_grid_missing) {
817 try {
818 if (fleet_edges.size() == 0) {
819 throw std::runtime_error(
820 "No resolvable fleet observation-bin geometry was available "
821 "to build the default biological size grid");
822 }
823
824 p->size_grid = fims_popdy::SizeGridBuilder::BuildDefaultFromFleetEdges(
825 fleet_edges);
826 EnforcePopulationGridNotCoarserThanFleetBins(p->size_grid, fleet_edges);
827
828 const std::size_t regular_bin_count =
829 p->size_grid.n_bins > 0 ? p->size_grid.n_bins - 1 : 0;
830
831 if (regular_bin_count > 200) {
832 FIMS_WARNING_LOG(
833 "Built-in default biological size grid created more than 200 "
834 "regular bins for population " +
835 fims::to_string(p->id) +
836 ". This may slow model evaluation. Consider providing a "
837 "biological size-grid override if needed.");
838 }
839
840 } catch (const std::exception& e) {
841 valid_model = false;
842 FIMS_ERROR_LOG(
843 "Failed to initialize default population size grid for "
844 "population " +
845 fims::to_string(p->id) + ": " + e.what());
846 }
847 return;
848 }
849
850 valid_model = false;
851 FIMS_ERROR_LOG(
852 "Population " + fims::to_string(p->id) +
853 " has an invalid biological size grid. Provide a consistent override "
854 "or leave the grid empty so FIMS can build the default.");
855 }
