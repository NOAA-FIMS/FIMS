58 virtual bool IsActive() const override {

94 // The current fixed matrix path does not vary by year, but year stays

102 // Stop if:
103 // - the fleet no longer exists
104 // - dimensions are invalid
105 // - the requested age is out of range
106 // - the fixed matrix does not have the expected size
107 if (fleet_ptr == nullptr || fleet_ptr->n_ages == 0 ||
108 fleet_ptr->n_lengths == 0 || age >= fleet_ptr->n_ages ||
109 fleet_ptr->age_to_length_conversion.size() !=
110 (fleet_ptr->n_ages * fleet_ptr->n_lengths)) {
111 return false;
112 }
113
114 // Resize the output row so it has one entry per fleet length bin.
115 out_row.resize(fleet_ptr->n_lengths);

118 // age x length matrix.
119 const size_t row_offset = age * fleet_ptr->n_lengths;
120
121 // Copy the fixed age-to-length probabilities for this age into out_row.
122 for (size_t l = 0; l < fleet_ptr->n_lengths; ++l) {
123 out_row[l] = fleet_ptr->age_to_length_conversion[row_offset + l];
124 }
