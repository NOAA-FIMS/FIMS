365 virtual void finalize() {
366 if (this->finalized) {
367 // log warning that finalize has been called more than once.
368 FIMS_WARNING_LOG("DnormDistribution " + fims::to_string(this->id_m) +
369 " has been finalized already.");
370 }
371
372 this->finalized = true; // indicate this has been called already
373
374 std::shared_ptr<fims_info::Information<double>> info =
375 fims_info::Information<double>::GetInstance();
376
377 fims_info::Information<double>::density_components_iterator it;
378
379 // search for density component in Information
380 it = info->density_components.find(this->id_m);
381 // if not found, just return
382 if (it == info->density_components.end()) {
383 FIMS_WARNING_LOG("DnormDistribution " + fims::to_string(this->id_m) +
384 " not found in Information.");
385 return;
386 } else {
387 std::shared_ptr<fims_distributions::NormalLPDF<double>> dnorm =
388 std::dynamic_pointer_cast<fims_distributions::NormalLPDF<double>>(
389 it->second);
390
391 this->lpdf_value = dnorm->lpdf;
392
393 size_t n_x = dnorm->get_n_x();
394
395 // If input log_sd is a scalar, resize to n_x and fill with the scalar
396 // value
397 if (this->log_sd.size() != n_x) {
398 // If log_sd size == 1 (scalar), repeat the entry
399 if (this->log_sd.size() == 1) {
400 auto tmp = this->log_sd[0]; // copy the one log_sd param
401 this->log_sd.resize(n_x);
402 for (size_t i = 0; i < n_x; ++i) {
403 this->log_sd[i] = tmp; // copies all fields in Param
404 }
405 } else {
406 // Handle error
407 FIMS_WARNING_LOG(
408 "log_sd size does not match number of observations and is not "
409 "scalar.");
410 }
411 }
412 for (size_t i = 0; i < n_x; i++) {
413 if (this->log_sd[i].estimation_type_m.get() == "constant") {
414 this->log_sd[i].final_value_m = this->log_sd[i].initial_value_m;
415 } else {
416 this->log_sd[i].final_value_m = dnorm->log_sd.get_force_scalar(i);
417 }
418 }
419
420 for (size_t i = 0; i < this->expected_mean.size(); i++) {
421 if (this->expected_mean[i].estimation_type_m.get() == "constant") {
422 this->expected_mean[i].final_value_m =
423 this->expected_mean[i].initial_value_m;
424 } else {
425 this->expected_mean[i].final_value_m = dnorm->expected_mean[i];
426 }
427 }
428
429 this->lpdf_vec = RealVector(n_x);
430 if (this->expected_values.size() == 1) {
431 this->expected_values.resize(n_x);
432 }
433 if (this->observed_values.size() == 1) {
434 this->observed_values.resize(n_x);
435 }
436
437 for (size_t i = 0; i < this->lpdf_vec.size(); i++) {
438 this->lpdf_vec[i] = dnorm->lpdf_vec[i];
439 this->expected_values[i].final_value_m = dnorm->get_expected(i);
440 this->observed_values[i].final_value_m = dnorm->get_observed(i);
441 }
442 }
443 }
