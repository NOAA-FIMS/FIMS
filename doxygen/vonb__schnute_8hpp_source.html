57 Type length_at_age(const Type& age) const {
58 if (reference_age_for_length_young > Type(0.0) &&
59 age < reference_age_for_length_young) {
60 // For early ages, avoid back-extrapolating the curved growth equation.
61 // This no-seasons ramp goes from length 0 at age 0 to length young at
62 // age young; seasonal growth may need a different transition in the
63 // future.
64 const Type age_nonnegative = age <= Type(0.0) ? Type(0.0) : age;
65 return mean_length_young * age_nonnegative /
66 reference_age_for_length_young;
67 }
68
69 const Type denom =
70 Type(1.0) -
71 fims_math::exp(-growth_coefficient * (reference_age_for_length_old -
72 reference_age_for_length_young));
73 // AD-safe floor to avoid divide-by-zero/NaN when denominator is tiny.
74 const Type denom_safe =
75 fims_math::ad_max(fims_math::ad_fabs(denom), static_cast<Type>(1e-8));
76 const Type numer =
77 Type(1.0) - fims_math::exp(-growth_coefficient *
78 (age - reference_age_for_length_young));
79 return mean_length_young +
80 (mean_length_old - mean_length_young) * numer / denom_safe;
81 }

105 void log_length_at_age_gradient(const Type& age,
106 Type& d_log_laa_d_length_young,
107 Type& d_log_laa_d_length_old,
108 Type& d_log_laa_d_k) const {
109 if (reference_age_for_length_young > Type(0.0) &&
110 age < reference_age_for_length_young) {
111 const Type age_nonnegative = age <= Type(0.0) ? Type(0.0) : age;
112 const Type ratio = age_nonnegative / reference_age_for_length_young;
113 const Type mean_length = mean_length_young * ratio;
114 const Type mean_length_safe =
115 fims_math::ad_max(mean_length, static_cast<Type>(1e-8));
116
117 d_log_laa_d_length_young = ratio / mean_length_safe;
118 d_log_laa_d_length_old = Type(0.0);
119 d_log_laa_d_k = Type(0.0);
120 return;
121 }

123 const Type age_delta_1 = age - reference_age_for_length_young;
124 const Type age_delta_2 =
125 reference_age_for_length_old - reference_age_for_length_young;
126
127 const Type exp_num = fims_math::exp(-growth_coefficient * age_delta_1);
128 const Type exp_den = fims_math::exp(-growth_coefficient * age_delta_2);
129
130 const Type numer = Type(1.0) - exp_num;
131 const Type denom = Type(1.0) - exp_den;
132 const Type denom_safe =
133 fims_math::ad_max(fims_math::ad_fabs(denom), static_cast<Type>(1e-8));
134
135 const Type ratio = numer / denom_safe;
136 const Type delta_length = mean_length_old - mean_length_young;
137 const Type mean_length = mean_length_young + delta_length * ratio;
138 const Type mean_length_safe =
139 fims_math::ad_max(mean_length, static_cast<Type>(1e-8));
140
141 const Type d_length_d_l1 = Type(1.0) - ratio;
142 const Type d_length_d_l2 = ratio;
143
144 const Type d_numer_d_k = age_delta_1 * exp_num;
145 const Type d_denom_d_k = age_delta_2 * exp_den;
146 const Type d_ratio_d_k = (d_numer_d_k * denom_safe - numer * d_denom_d_k) /
147 (denom_safe * denom_safe);
148 const Type d_length_d_k = delta_length * d_ratio_d_k;
149
150 d_log_laa_d_length_young = d_length_d_l1 / mean_length_safe;
151 d_log_laa_d_length_old = d_length_d_l2 / mean_length_safe;
152 d_log_laa_d_k = d_length_d_k / mean_length_safe;
153 }
