186 bool TryFinalizeMappedProbabilityRow(
187 fims::Vector<Type>& row, std::size_t expected_size,
188 const Type& minimum_bin_prob = static_cast<Type>(1e-12)) const {
189 if (row.size() != expected_size) {
190 return false;
191 }
192
193 Type row_sum = static_cast<Type>(0.0);
194 for (std::size_t i = 0; i < row.size(); ++i) {
195 row[i] += minimum_bin_prob;
196 row_sum += row[i];
197 }
198
199 for (std::size_t i = 0; i < row.size(); ++i) {
200 row[i] /= row_sum;
201 }
202
203 return true;
204 }

213 fims::Vector<Type> BuildMappedFleetAgeToLengthConversionRow(
214 std::size_t year_index, std::size_t age_index) const {
215 std::shared_ptr<Fleet<Type>> fleet_ptr = fleet_.lock();
216 std::shared_ptr<SizeDistributionProviderBase<Type>> size_provider_ptr =
217 TryGetSizeProvider();
218
219 if (fleet_ptr == nullptr || size_provider_ptr == nullptr) {
220 return fims::Vector<Type>();
221 }
222
223 const SizeGrid* population_size_grid = size_provider_ptr->TryGetSizeGrid();
224 if (population_size_grid == nullptr ||
225 !population_size_grid->IsConsistent() ||
226 population_size_grid->n_bins == 0) {
227 return fims::Vector<Type>();
228 }
229
230 if (fleet_ptr->n_years == 0 || fleet_ptr->n_lengths == 0 ||
231 fleet_ptr->lengths.size() != fleet_ptr->n_lengths ||
232 fleet_ptr->length_bin_edges.size() != fleet_ptr->n_lengths + 1 ||
233 !SizeBinMapping::HasStrictlyIncreasingEdges(
234 fleet_ptr->length_bin_edges) ||
235 year_index >= fleet_ptr->n_years || age_index >= fleet_ptr->n_ages) {
236 return fims::Vector<Type>();
237 }
238
239 const fims::Vector<Type> population_prob_size =
240 BuildPopulationProbSizeRow(year_index, age_index);
241 if (population_prob_size.size() != population_size_grid->n_bins) {
242 return fims::Vector<Type>();
243 }
244
245 const fims::Vector<double> mapping_fleet_edges =
246 SizeBinMapping::ExpandDestinationEdgesToCoverSourceRange(
247 population_size_grid->edges, fleet_ptr->length_bin_edges);
248
249 const fims::Vector<fims::Vector<double>> rebin_weights =
250 SizeBinMapping::BuildRebinWeights(population_size_grid->edges,
251 mapping_fleet_edges);
252
253 fims::Vector<Type> fleet_row =
254 SizeBinMapping::ApplyRebinWeights(rebin_weights, population_prob_size);
255
256 if (!TryFinalizeMappedProbabilityRow(fleet_row, fleet_ptr->n_lengths)) {
257 throw std::runtime_error(
258 "AgeToLengthConversionDerived produced an invalid mapped fleet "
259 "probability row");
260 }
261
262 return fleet_row;
263 }
