67 void PrepareSizeProducts() override {
68 if (!population_size_grid_) {
69 throw std::runtime_error(
70 "GrowthDerivedSizeProvider requires a population biological size "
71 "grid");
72 }
73
74 if (!population_size_grid_->IsConsistent()) {
75 throw std::runtime_error(
76 "GrowthDerivedSizeProvider requires a consistent population "
77 "biological size grid");
78 }
79
80 if (population_size_grid_->n_bins == 0) {
81 throw std::runtime_error(
82 "GrowthDerivedSizeProvider requires a population biological size grid"
83 " with at least one bin");
84 }
85
86 if (n_years_ == 0 || n_ages_ == 0) {
87 throw std::runtime_error(
88 "GrowthDerivedSizeProvider requires nonzero population dimensions");
89 }
90
91 const GrowthProducts<Type>& growth_products = PreparedGrowthProducts();
92
93 if (growth_products.n_years != n_years_ ||
94 growth_products.n_ages != n_ages_) {
95 throw std::runtime_error(
96 "GrowthDerivedSizeProvider growth products do not match "
97 "configured population dimensions");
98 }
99
100 if (growth_products.n_sexes != 1) {
101 throw std::runtime_error(
102 "GrowthDerivedSizeProvider currently requires pooled growth "
103 "products with n_sexes == 1");
104 }
105
106 size_products_.ResizeProbSizeOnly(n_years_, n_ages_,
107 population_size_grid_->n_bins);
108
109 const Type plus_group_warning_threshold = static_cast<Type>(0.01);
110 bool plus_group_warning_needed = false;
111 std::size_t warning_year_index = 0;
112 std::size_t warning_age_index = 0;
113
114 for (std::size_t year_index = 0; year_index < n_years_; ++year_index) {
115 for (std::size_t age_index = 0; age_index < n_ages_; ++age_index) {
116 fims::Vector<Type> prob_row(population_size_grid_->n_bins);
117 Type row_sum = static_cast<Type>(0.0);
118 const Type minimum_bin_prob = static_cast<Type>(1e-12);
119
120 for (std::size_t size_bin_index = 0;
121 size_bin_index < population_size_grid_->n_bins; ++size_bin_index) {
122 Type bin_prob = PopulationSizeBinProb(growth_products, year_index,
123 age_index, size_bin_index) +
124 minimum_bin_prob;
125
126 prob_row[size_bin_index] = bin_prob;
127 row_sum += bin_prob;
128 }
129
130 for (std::size_t size_bin_index = 0;
131 size_bin_index < population_size_grid_->n_bins; ++size_bin_index) {
132 size_products_.ProbSize(year_index, age_index, size_bin_index) =
133 prob_row[size_bin_index] / row_sum;
134 }
135
136 const Type plus_group_prob = size_products_.ProbSize(
137 year_index, age_index, population_size_grid_->n_bins - 1);
138
139 if (!plus_group_warning_needed &&
140 plus_group_prob > plus_group_warning_threshold) {
141 plus_group_warning_needed = true;
142 warning_year_index = year_index;
143 warning_age_index = age_index;
144 }
145 }
146 }
147
148 if (plus_group_warning_needed && !plus_group_warning_emitted_) {
149 FIMS_WARNING_LOG(
150 "Growth-derived size preparation placed more than 0.01 "
151 "probability in the terminal biological plus-group bin for at "
152 "least one year-age row. First occurrence was year index " +
153 fims::to_string(warning_year_index) + ", age index " +
154 fims::to_string(warning_age_index) +
155 ". The population biological size grid may need to be widened.");
156 plus_group_warning_emitted_ = true;
157 }
158
159 // Weight depends only on length, so it is evaluated once per bin. The
160 // first and last bins are weighted at their centers even though they hold
161 // the lower and upper tails, as in SS3.
162 fims::Vector<Type> weight_at_bin(population_size_grid_->n_bins);
163 for (std::size_t size_bin_index = 0;
164 size_bin_index < population_size_grid_->n_bins; ++size_bin_index) {
165 weight_at_bin[size_bin_index] =
166 growth_observation_->EvaluateWeightAtLength(static_cast<Type>(
167 population_size_grid_->centers[size_bin_index]));
168 }
169
170 mean_weight_at_age_.resize(n_years_ * n_ages_);
171 for (std::size_t year_index = 0; year_index < n_years_; ++year_index) {
172 for (std::size_t age_index = 0; age_index < n_ages_; ++age_index) {
173 Type mean_weight = static_cast<Type>(0.0);
174 for (std::size_t size_bin_index = 0;
175 size_bin_index < population_size_grid_->n_bins; ++size_bin_index) {
176 mean_weight +=
177 size_products_.ProbSize(year_index, age_index, size_bin_index) *
178 weight_at_bin[size_bin_index];
179 }
180 mean_weight_at_age_[year_index * n_ages_ + age_index] = mean_weight;
181 }
182 }
183
184 size_products_prepared_ = true;
185 }

256 Type PopulationSizeBinProb(const GrowthProducts<Type>& growth_products,
257 std::size_t year_index, std::size_t age_index,
258 std::size_t size_bin_index) const {
259 if (!population_size_grid_ || population_size_grid_->n_bins == 0) {
260 throw std::runtime_error(
261 "GrowthDerivedSizeProvider requires a population biological size grid"
262 " before computing ProbSize");
263 }
264
265 const Type mean_laa = growth_products.MeanLAA(year_index, age_index, 0);
266 const Type sd_laa =
267 fims_math::ad_max(growth_products.SdLAA(year_index, age_index, 0),
268 static_cast<Type>(1e-8));
269
270 if (population_size_grid_->n_bins == 1) {
271 return static_cast<Type>(1.0);
272 }
273
274 if (size_bin_index == 0) {
275 const Type upper = static_cast<Type>(population_size_grid_->edges[1]);
276 return fims_math::pnorm(upper, mean_laa, sd_laa);
277 }
278
279 if (size_bin_index + 1 == population_size_grid_->n_bins) {
280 const Type lower = static_cast<Type>(
281 population_size_grid_->edges[population_size_grid_->n_bins - 1]);
282 return fims_math::pnorm(static_cast<Type>(2.0) * mean_laa - lower,
283 mean_laa, sd_laa);
284 }
285
286 const Type lower =
287 static_cast<Type>(population_size_grid_->edges[size_bin_index]);
288 const Type upper =
289 static_cast<Type>(population_size_grid_->edges[size_bin_index + 1]);
290
291 return fims_math::pnorm(upper, mean_laa, sd_laa) -
292 fims_math::pnorm(lower, mean_laa, sd_laa);
293 }
