355 virtual const Type evaluate(int year, const double& a) override {
356 if (a < 0.0) {
357 throw std::runtime_error("Negative age not supported");
358 }
359 const double a_round = std::round(a);
360 const double tol = 1e-8;
361 if (std::fabs(a - a_round) > tol) {
362 throw std::runtime_error("Non-integer age not supported yet");
363 }
364 EnsureParamsSet();
365 const Type ref_age_young = CurrentReferenceAgeForLengthYoung();
366 const Type ref_age_old = CurrentReferenceAgeForLengthOld();
367 if (ref_age_old <= ref_age_young) {
368 throw std::runtime_error(
369 "VonBertalanffySchnuteGrowth reference_age_for_length_old must be > "
370 "reference_age_for_length_young");
371 }
372
373 if (!model_) {
374 return EvaluateWithFunctor(year, a);
375 }
376
377 SyncParamsToModel();
378 model_->Prepare();
379 const auto& p = model_->GetProducts();
380 const double offset = age_offset_set_ ? age_offset_ : 0.0;
381 const double age_index_raw = a_round - offset;
382 const double age_index_round = std::round(age_index_raw);
383 if (std::fabs(age_index_raw - age_index_round) <= tol &&
384 age_index_round >= 0.0) {
385 const std::size_t age_index = static_cast<std::size_t>(age_index_round);
386 if (age_index < p.n_ages) {
387 const std::size_t year_index =
388 (year >= 0 && static_cast<std::size_t>(year) < p.n_years)
389 ? static_cast<std::size_t>(year)
390 : static_cast<std::size_t>(0);
391 return p.MeanWAA(year_index, age_index, 0);
392 }
393 }
394
395 // Outside cached model age bins: fall back to direct functor evaluation.
396 return EvaluateWithFunctor(year, a);
397 }

672 void EnsureParamsSet() const {
673 if (!use_param_vectors_ || mean_length_young_vector_.size() < 1 ||
674 mean_length_old_vector_.size() < 1 ||
675 growth_coefficient_vector_.size() < 1 ||
676 reference_age_for_length_young_vector_.size() < 1 ||
677 reference_age_for_length_old_vector_.size() < 1 ||
678 length_weight_a_vector_.size() < 1 ||
679 length_weight_b_vector_.size() < 1) {
680 throw std::runtime_error(
681 "VonBertalanffySchnuteGrowth parameters not set");
682 }
683
684 if (mean_length_young_vector_.size() != 1 ||
685 mean_length_old_vector_.size() != 1 ||
686 growth_coefficient_vector_.size() != 1 ||
687 reference_age_for_length_young_vector_.size() != 1 ||
688 reference_age_for_length_old_vector_.size() != 1 ||
689 length_weight_a_vector_.size() != 1 ||
690 length_weight_b_vector_.size() != 1) {
691 throw std::runtime_error(
692 "VonBertalanffySchnuteGrowthModelAdapter currently supports a single "
693 "growth pattern; expected size 1 for von Bertalanffy--Schnute and "
694 "length-weight parameter vectors");
695 }
696
697 const bool has_sd = HasInterpolationSdInputs();
698 const bool has_any_structured_delta = HasAnyStructuredDeltaMethodInput();
699 const bool has_structured_delta = HasStructuredDeltaMethodInputs();
700
701 if (!has_sd && !has_any_structured_delta) {
702 throw std::runtime_error(
703 "VonBertalanffySchnuteGrowthModelAdapter requires either "
704 "both length_at_age_sd_young and length_at_age_sd_old, "
705 "or the structured delta-method growth variability inputs");
706 }
707
708 if (has_any_structured_delta && !has_structured_delta) {
709 throw std::runtime_error(
710 "VonBertalanffySchnuteGrowthModelAdapter requires all six structured "
711 "delta-method variability inputs when using that path");
712 }
713
714 if (has_sd && has_structured_delta) {
715 throw std::runtime_error(
716 "von Bertalanffy--Schnute growth adapter requires variability inputs "
717 "for exactly one supported path. Supply either the interpolation "
718 "inputs length_at_age_sd_young and length_at_age_sd_old or the full "
719 "delta-method variability inputs, but not both");
720 }
721
722 if (has_sd && (length_at_age_sd_young_vector_.size() != 1 ||
723 length_at_age_sd_old_vector_.size() != 1)) {
724 throw std::runtime_error(
725 "VonBertalanffySchnuteGrowthModelAdapter requires exactly one "
726 "length_at_age_sd_young value and one length_at_age_sd_old value");
727 }
728
729 if (has_structured_delta &&
730 (log_sd_mean_length_young_vector_.size() != 1 ||
731 log_sd_mean_length_old_vector_.size() != 1 ||
732 log_sd_growth_coefficient_vector_.size() != 1 ||
733 mean_length_young_mean_length_old_logit_corr_vector_.size() != 1 ||
734 mean_length_young_growth_coefficient_logit_corr_vector_.size() != 1 ||
735 mean_length_old_growth_coefficient_logit_corr_vector_.size() != 1)) {
736 throw std::runtime_error(
737 "VonBertalanffySchnuteGrowthModelAdapter currently supports a single "
738 "structured delta-method variability parameter set; expected "
739 "size 1 for each structured uncertainty input");
740 }
