322inline const Type double_logistic(const Type &inflection_point_asc,
323 const Type &slope_asc,

326 return (static_cast<Type>(1.0)) /

352template <class Type>
353const Type ad_fabs(const Type &x, Type C = 1e-5) {
354 // C = 0 gives the exact |x|, which has a NaN derivative at x = 0. C < 0
355 // gives NaN when |x| is small. C is a fixed setting, not an estimated
356 // value, so checking it once when TMB records the tape is enough.
357 if (fims_math::Value(C) <= 0.0) {
358 throw std::invalid_argument("fims_math::ad_fabs: C is " +
359 fims::to_string(fims_math::Value(C)) +
360 ", which is not greater than 0.");
361 }
362 return sqrt((x * x) + C);
363}

435template <typename Type>
436inline fims::Vector<Type> apply_ageing_error(
437 const fims::Vector<Type> &proportion_at_true_age,
438 const fims::Vector<Type> &ageing_error, size_t year = 0) {
439 const size_t n_ages = proportion_at_true_age.size();
440 const size_t matrix_size = n_ages * n_ages;
441 const size_t matrix_offset =
442 (ageing_error.size() == matrix_size) ? 0 : year * matrix_size;
443
444 fims::Vector<Type> proportion_at_observed_age(n_ages, static_cast<Type>(0.0));
445 for (size_t true_age = 0; true_age < n_ages; true_age++) {
446 for (size_t observed_age = 0; observed_age < n_ages; observed_age++) {
447 proportion_at_observed_age[observed_age] +=
448 proportion_at_true_age[true_age] *
449 ageing_error[matrix_offset + true_age * n_ages + observed_age];
450 }
451 }
452 return proportion_at_observed_age;
453}
