300template <class Type>
301inline const Type double_logistic(const Type &inflection_point_asc,
302 const Type &slope_asc,
303 const Type &inflection_point_desc,
304 const Type &slope_desc, const Type &x) {
305 return (static_cast<Type>(1.0)) /
306 (static_cast<Type>(1.0) +
307 exp(Type(-1.0) * slope_asc * (x - inflection_point_asc))) *
308 (static_cast<Type>(1.0) -
309 (static_cast<Type>(1.0)) /
310 (static_cast<Type>(1.0) +
311 exp(Type(-1.0) * slope_desc * (x - inflection_point_desc))));
312}
