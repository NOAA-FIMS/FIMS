104template <typename Type>
105void EnsureAgeToLengthConversionFleet(
106 const std::shared_ptr<Population<Type>>& population,
107 const std::shared_ptr<Fleet<Type>>& fleet) {
108 if (fleet == nullptr) {
109 throw std::runtime_error(
110 "Fleet pointer was null while resolving age-to-length conversion.");
111 }
112
113 if (population == nullptr) {
114 throw std::runtime_error(
115 "Population pointer was null while resolving age-to-length "
116 "conversion.");
117 }
118
119 if (!fleet->requires_age_length_mapping || fleet->n_lengths == 0) {
120 return;
121 }
122
123 std::shared_ptr<GrowthDerivedObservationBase<Type>> growth_observation =
124 std::dynamic_pointer_cast<GrowthDerivedObservationBase<Type>>(
125 population->growth);
126
127 if (fleet->age_to_length_conversion_model != nullptr) {
128 if (growth_observation != nullptr) {
129 std::shared_ptr<AgeToLengthConversionDerived<Type>>
130 age_to_length_conversion_derived =
131 std::dynamic_pointer_cast<AgeToLengthConversionDerived<Type>>(
132 fleet->age_to_length_conversion_model);
133
134 if (age_to_length_conversion_derived != nullptr &&
135 age_to_length_conversion_derived->IsActive() &&
136 age_to_length_conversion_derived->PrepareForCurrentState()) {
137 return;
138 }
139 } else if (fleet->age_to_length_conversion_model->IsActive() &&
140 fleet->age_to_length_conversion_model
141 ->PrepareForCurrentState()) {
142 return;
143 }
144 }
145
146 fleet->age_to_length_conversion_model =
147 BuildAgeToLengthConversionFleet<Type>(population, fleet);
148
149 if (fleet->age_to_length_conversion_model != nullptr) {
150 return;
151 }
152
153 std::stringstream ss;
154 ss << "Fleet " << fleet->GetId()
155 << " has length composition bins but no usable age-to-length conversion "
156 "path.";
157
158 if (growth_observation != nullptr) {
159 ss << " This population uses a growth-derived-capable growth object, so "
160 << "the growth-derived age-to-length conversion path with population "
161 "size-provider support "
162 << "was required and could not be built or prepared.";
163 } else {
164 ss << " Provide a valid fixed age-to-length conversion matrix for this "
165 << "fleet or use a supported growth-derived-capable growth object.";
166 }
167
168 FIMS_ERROR_LOG(ss.str());
169 throw std::runtime_error(ss.str());
170}
